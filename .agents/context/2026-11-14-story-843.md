Query:

```sql
WITH deleted AS (
 SELECT m.* FROM media_items m
 WHERE EXISTS (SELECT 1 FROM trash_deleted_releases d WHERE d.arr_kind=m.arr_kind
   AND (d.media_item_id=m.id OR (m.arr_kind='radarr' AND d.tmdb_id=m.tmdb_id)
     OR (m.arr_kind='sonarr' AND d.tvdb_id=m.tvdb_id)))
 OR EXISTS (SELECT 1 FROM ledger_events le WHERE le.media_item_id=m.id AND le.event_type='trash_expedited')
 OR EXISTS (SELECT 1 FROM trash_batch_items bi WHERE bi.media_item_id=m.id AND bi.state='deleted')
)
SELECT arr_kind, count(*) AS trash_deleted_titles,
 count(*) FILTER (WHERE deleted_from_arr_at IS NOT NULL) AS tombstoned,
 count(*) FILTER (WHERE deleted_from_arr_at IS NOT NULL OR NOT monitored) AS ledger_bulk_actionable,
 count(*) FILTER (WHERE monitored AND deleted_from_arr_at IS NOT NULL) AS failsafe_candidates
FROM deleted GROUP BY arr_kind ORDER BY arr_kind;
```
